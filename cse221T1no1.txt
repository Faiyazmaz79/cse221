import java.util.Scanner;
 public class oddeven {
    public static void main(String [] args){
        Scanner sc = new Scanner(System.in);
 
        int num = sc.nextInt();
 
        for (int i = 0; i < num; i++){
            int n = sc.nextInt();
 
        
 
        if (n%2==0) {
            System.out.println(n + " is an Even number.");
        }
        else {
            System.out.println(n + " is an Odd number.");
        }
     }
 
 
    }
}
