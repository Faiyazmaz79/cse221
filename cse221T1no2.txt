import java.util.*;
 
public class oPeration {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
 
        int n = sc.nextInt();   
 
        for (int i = 0; i < n; i++) {
            
 
            String word = sc.next();  
            double a = sc.nextDouble(); 
            String op = sc.next();      
            double b = sc.nextDouble(); 
 
            double answer = 0;
 
            
            if (op.equals("+")) {
                answer = a + b;
            }
            else if (op.equals("-")) {
                answer = a - b;
            }
            else if (op.equals("*")) {
                answer = a * b; 
            }
            else if (op.equals("/")) {
                answer = a / b;
            }
            
 
          
            System.out.printf("%.6f%n", answer);
        }
 
        sc.close();
    }
 
 
    
}
